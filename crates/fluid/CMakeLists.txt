cmake_minimum_required(VERSION 3.24)
project(LumaFluid VERSION 0.10.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Widgets Svg Multimedia ShaderTools Network)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(luma-mail-fluid WIN32 src/main.cpp src/bridge.cpp src/bridge.h src/motion.cpp src/motion.h src/motion_smoke.cpp src/motion_smoke.h src/rich_document.cpp src/rich_document.h src/composer_smoke.cpp src/clipboard.cpp src/profiles.cpp src/profile_images.cpp src/profiles.h src/profile_smoke.cpp src/html_surface.cpp src/html_surface.h src/reader_profile.cpp src/reader_profile.h)
target_sources(luma-mail-fluid PRIVATE src/html_smoke.cpp src/runtime_cleanup.cpp src/runtime_cleanup.h)
target_sources(luma-mail-fluid PRIVATE src/read_document.cpp src/read_document.h)
target_sources(luma-mail-fluid PRIVATE src/record_model.cpp src/record_model.h)
target_sources(luma-mail-fluid PRIVATE src/sync_smoke.cpp src/actions.cpp src/views.cpp src/navigation_smoke.cpp src/search_smoke.cpp src/conflict_smoke.cpp)
target_sources(luma-mail-fluid PRIVATE src/attachment_preview.cpp src/attachment_preview.h src/attachment_thumbnails.cpp)
target_sources(luma-mail-fluid PRIVATE src/attachment_smoke.cpp)
target_sources(luma-mail-fluid PRIVATE src/single_instance.cpp src/single_instance.h)
target_sources(luma-mail-fluid PRIVATE src/resource_smoke.cpp)
target_sources(luma-mail-fluid PRIVATE src/signature_smoke.cpp)
target_include_directories(luma-mail-fluid PRIVATE src)
target_link_libraries(luma-mail-fluid PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Svg Qt6::Multimedia Qt6::Network)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
if(WIN32)
    set(LUMA_PLATFORM_READER qml/WindowsReader.qml)
else()
    find_package(Qt6 6.8 REQUIRED COMPONENTS WebEngineQuick WebEngineCore)
    target_link_libraries(luma-mail-fluid PRIVATE Qt6::WebEngineQuick Qt6::WebEngineCore)
    target_compile_definitions(luma-mail-fluid PRIVATE LUMA_WEBENGINE)
    set(LUMA_PLATFORM_READER qml/LinuxReader.qml)
endif()
qt_add_qml_module(luma-mail-fluid URI Luma.Fluid VERSION 1.0
    QML_FILES qml/Main.qml qml/Theme.qml qml/FluidColor.qml qml/FluidButton.qml qml/FluidCombo.qml qml/FluidCheck.qml qml/Icon.qml
        qml/Avatar.qml qml/Sidebar.qml qml/NavigationEntry.qml qml/Inbox.qml qml/Reader.qml
        qml/Composer.qml qml/RecipientField.qml qml/RowMorph.qml qml/ActivityPill.qml qml/SettingsSheet.qml qml/AttachmentTile.qml qml/AttachmentPane.qml qml/DownloadPill.qml qml/FluidMenu.qml qml/LabelSection.qml qml/LabelDialog.qml qml/MailActionsMenu.qml qml/ConfirmSheet.qml qml/FluidField.qml qml/SearchSheet.qml qml/DraftCompareSheet.qml qml/MessageActionsMenu.qml qml/PaneHandle.qml qml/SignatureSheet.qml ${LUMA_PLATFORM_READER})
file(GLOB LUMA_ICONS CONFIGURE_DEPENDS "../../data/icons/scalable/actions/*.svg")
qt_add_resources(luma-mail-fluid "icons" PREFIX "/icons" BASE "../../data/icons/scalable/actions" FILES ${LUMA_ICONS})
qt_add_resources(luma-mail-fluid "brand" PREFIX "/brand" BASE "../../data" FILES "../../data/luma-wordmark.png" "../../data/io.github.lumamail.Luma.svg")
qt_add_shaders(luma-mail-fluid "icon_shaders" PREFIX "/" FILES shaders/icon.frag)
if(UNIX AND NOT APPLE)
    find_package(Qt6 REQUIRED COMPONENTS DBus)
    target_link_libraries(luma-mail-fluid PRIVATE Qt6::DBus)
    target_sources(luma-mail-fluid PRIVATE src/notifications_linux.cpp src/notifications_linux.h)
endif()
install(TARGETS luma-mail-fluid RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(instance-test tests/instance_test.cpp src/single_instance.cpp src/single_instance.h)
    target_include_directories(instance-test PRIVATE src)
    target_link_libraries(instance-test PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME single-instance COMMAND instance-test)
    qt_add_executable(motion-test tests/motion_test.cpp src/motion.cpp src/motion.h)
    target_include_directories(motion-test PRIVATE src)
    target_link_libraries(motion-test PRIVATE Qt6::Core Qt6::Qml Qt6::Test)
    add_test(NAME motion COMMAND motion-test)
    qt_add_executable(document-test tests/document_test.cpp src/rich_document.cpp src/rich_document.h src/read_document.cpp src/read_document.h)
    target_include_directories(document-test PRIVATE src)
    target_link_libraries(document-test PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Test)
    add_test(NAME rich-document COMMAND document-test)
    set_tests_properties(rich-document PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(profile-test tests/profile_test.cpp src/profile_images.cpp src/profiles.h)
    target_include_directories(profile-test PRIVATE src)
    target_link_libraries(profile-test PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Test)
    add_test(NAME profile-images COMMAND profile-test)
    set_tests_properties(profile-images PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(runtime-test tests/runtime_test.cpp src/runtime_cleanup.cpp src/runtime_cleanup.h)
    target_include_directories(runtime-test PRIVATE src)
    target_link_libraries(runtime-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME reader-runtime COMMAND runtime-test)
    qt_add_executable(model-test tests/model_test.cpp src/record_model.cpp src/record_model.h)
    target_include_directories(model-test PRIVATE src)
    target_link_libraries(model-test PRIVATE Qt6::Core Qt6::Test)
    add_test(NAME incremental-inbox-model COMMAND model-test)
    if(UNIX AND NOT APPLE)
        find_program(LUMA_DBUS_RUN_SESSION dbus-run-session REQUIRED)
        qt_add_executable(notification-test tests/notifications_test.cpp src/notifications_linux.cpp src/notifications_linux.h)
        target_include_directories(notification-test PRIVATE src)
        target_link_libraries(notification-test PRIVATE Qt6::Core Qt6::DBus Qt6::Test)
        add_test(NAME system-notification-daemon COMMAND ${LUMA_DBUS_RUN_SESSION} -- $<TARGET_FILE:notification-test>)
    endif()
endif()
